rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn(){ return request.auth != null; }
    function isAdmin(){ return signedIn() && request.auth.uid == "5dlGX6JlrUQHyjFWSHB9Dye0r1E3"; }
    function clientDoc(clientId){ return get(/databases/$(database)/documents/clients/$(clientId)); }
    function ownsClient(clientId){ return signedIn() && (clientDoc(clientId).data.authUid == request.auth.uid || clientDoc(clientId).data.email == request.auth.token.email); }

    match /clients/{clientId} {
      allow read, write: if isAdmin();
      allow read: if signedIn() && (resource.data.authUid == request.auth.uid || resource.data.email == request.auth.token.email);
      allow update: if signedIn() && resource.data.email == request.auth.token.email
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['authUid','firstAccessAt'])
        && request.resource.data.authUid == request.auth.uid;
    }
    match /trips/{id} { allow read, write: if isAdmin(); allow read: if signedIn() && resource.data.published == true && ownsClient(resource.data.clientId); }
    match /payments/{id} { allow read, write: if isAdmin(); allow read: if signedIn() && ownsClient(resource.data.clientId); }
    match /paymentPlans/{id} { allow read, write: if isAdmin(); allow read: if signedIn() && ownsClient(resource.data.clientId); }
    match /tripDocuments/{id} { allow read, write: if isAdmin(); allow read: if signedIn() && ownsClient(resource.data.clientId); }
    match /memories/{id} { allow read, write: if isAdmin(); allow read, create, update: if signedIn() && ownsClient(request.resource.data.clientId); }
    match /recommendations/{id} { allow read, write: if isAdmin(); allow read: if signedIn(); }
    match /conciergeRequests/{id} { allow read, write: if isAdmin(); allow create, read: if signedIn() && ownsClient(request.resource.data.clientId); }
    match /benefits/{id} { allow read, write: if isAdmin(); allow read: if signedIn(); }
    match /itineraryTemplates/{id} { allow read, write: if isAdmin(); }
    match /users/{id} { allow read, write: if isAdmin(); allow read: if signedIn() && request.auth.uid == id; }
    match /settings/{document=**} { allow read, write: if isAdmin(); }
    match /{document=**} { allow read, write: if false; }
  }
}
